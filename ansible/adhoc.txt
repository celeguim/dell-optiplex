# ping all
ANSIBLE_CONFIG=ansible.cfg ansible all -m ping -i inventory.yaml

# check date/time
ANSIBLE_CONFIG=ansible.cfg ansible all -i inventory.yaml -m ansible.builtin.shell --args 'hostname; date; df -kh'

# yum update / root
ANSIBLE_CONFIG=ansible.cfg ansible all -i inventory.yaml -m ansible.builtin.shell \
--args 'apt update && apt -y upgrade && apt autoclean' --become

# service status / root
ANSIBLE_CONFIG=ansible.cfg ansible all -i inventory.yaml -m ansible.builtin.service --args 'name=httpd state=started' --become

# install/configure apache
ANSIBLE_CONFIG=ansible.cfg ansible-playbook -i inventory.yaml apache2.yaml --limit dell-optiplex

# configurar rede estatica
ANSIBLE_CONFIG=ansible.cfg ansible-playbook -i inventory.yaml playbooks/netplan.yaml --limit dell-optiplex --become -vvv
